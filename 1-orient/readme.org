*** Create AGP files from paf
[[https://jcottonres.biomedcentral.com/articles/10.1186/s42397-018-0002-1][Wang, Wendel, and Hua 2018]] prefix

#+header: :var genomes=../0-ref/readme.org:genomes[,0]
#+begin_src sh :tangle 2-infer-agp/from-paf.sh
ROOT=$(git rev-parse --show-toplevel)

for name in ${genomes[@]}; do
    perl $ROOT/apps/agp-from-paf.pl $DIR/1-align/$name.paf |
        awk '$6 ~ /^[AD][01][0-9]/ {
            $1=gensub("([AD])([01][0-9]).*", "\\1h_\\2", "g", $6)
            print; }' OFS="\t" |
        sort -k1,1 > $DIR/2-infer-agp/$name.inferred.agp
done
#+end_src
*** Verify Inferred AGP files

Keeping the longest 26 (number of chrs) fixes the agp files
#+header: :var genomes=../0-ref/readme.org:genomes[,0]
#+begin_src shell :tangle 3-curate-agp/run.sh :exports both
for name in ${genomes[@]}; do
    sort -k3,3nr $DIR/2-infer-agp/$name.inferred.agp |
        head -n26 |
        sort -k1,1 > $DIR/3-curate-agp/$name.agp
done
#+end_src


*** Create fasta files
#+header: :var genomes=../0-ref/readme.org:genomes[,0]
#+begin_src shell :tangle 4-genomes/run.sh
ROOT=$(git rev-parse --show-toplevel)
module load samtools/1.19.2

PATH=$HOME/.local/bin:$PATH
PYTHONPATH=$HOME/.local/lib/python3.9/site-packages/

name=${genomes[$SLURM_ARRAY_TASK_ID]}

AGP_FILE=$DIR/3-curate-agp/$name.agp
if [[ ! -f $AGP_FILE ]] ; then
    echo "Can't find $name agp file";
    exit
fi

cut -f 6 $AGP_FILE |
    samtools faidx -r - $ROOT/0-ref/$name.fa.gz \
        > $TMPDIR/$name.fa

python3 $HOME/.local/bin/agptools assemble \
    $TMPDIR/$name.fa $AGP_FILE |
    bgzip > $DIR/4-genomes/$name.fa.gz

#+end_src
