for name in "${filter[@]}"; do
    unset genomes[$name]
done

for name in "${genomes[@]}"; do
    out=${name//-/_}
    tmp=$(mktemp -d)

    zcat $ROOT/1-orient/4-genomes/$name.fa.gz > $tmp/$out.fa
    zcat $ROOT/2-first-pass/7-rename/$name.gtf.gz |
        awk '$3 == "transcript" {
        $9 = sprintf("transcript_id \"%s\"", $9)} 1' \
            FS="\t" OFS="\t" > $tmp/$out.gtf

    gffread -J -y $tmp/$out.aa.fa \
        -g $tmp/$out.fa $tmp/$out.gtf
    rm $tmp/$out.fa{,.fai}
    samtools faidx $tmp/$out.aa.fa

    awk '$3 == "transcript" && $6 >= 0.75 {
              print $1,$4,$5,
                    gensub("transcript_id \"(.*)\"",
                           "\\1", "g", $9) }' \
              FS="\t" OFS="\t" \
              $tmp/$out.gtf > $DIR/1-whole/bed/$out.bed

    rm $tmp/$out.gtf

    comm -12 <( cut -f 4 $DIR/1-whole/bed/$out.bed | sort ) \
        <(cut -f 1 $tmp/$out.aa.fa.fai | sort ) |
        samtools faidx -c -r - $tmp/$out.aa.fa \
            > $DIR/1-whole/peptide/$out.fa

    rm $tmp/$out.aa.fa{,.fai}
    rmdir $tmp
done

    unset genomes[$name]
done

for name in "${genomes[@]}"; do
    out=${name//-/_}
    tmp=$(mktemp -d)

    zcat $ROOT/1-orient/4-genomes/$name.fa.gz > $tmp/$out.fa
    zcat $ROOT/2-first-pass/7-rename/$name.gtf.gz |
        awk '$3 == "transcript" {
        $9 = sprintf("transcript_id \"%s\"", $9)} 1' \
            FS="\t" OFS="\t" > $tmp/$out.gtf

    gffread -J -y $tmp/$out.aa.fa \
        -g $tmp/$out.fa $tmp/$out.gtf
    rm $tmp/$out.fa{,.fai}
    samtools faidx $tmp/$out.aa.fa

    prefix=("$out.At" "$out.Dt")
    for sub in A D; do
        grep "^${sub}h" $tmp/$out.gtf \
            > $tmp/$out.${sub}t.gtf
    done
    rm $tmp/$out.gtf

    for pre in ${prefix[@]}; do
        awk '$3 == "transcript" && $6 >= 0.75 {
              print $1,$4,$5,
                    gensub("transcript_id \"(.*)\"",
                           "\\1", "g", $9) }' \
                  FS="\t" OFS="\t" \
                  $tmp/$pre.gtf > $DIR/2-split/bed/$pre.bed
        rm $tmp/$pre.gtf

        comm -12 <( cut -f 4 $DIR/2-split/bed/$pre.bed | sort ) \
            <(cut -f 1 $tmp/$out.aa.fa.fai | sort ) |
            samtools faidx -c -r - $tmp/$out.aa.fa \
                > $DIR/2-split/peptide/$pre.fa
    done

    rm $tmp/$out.aa.fa{,.fai}
    rmdir $tmp
done
