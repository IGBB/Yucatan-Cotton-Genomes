export NCBI_FCS_REPORT_ANALYTICS=0

for name in "${genomes[@]}"; do
    zcat $ROOT/1-orient/4-genomes/$name.fa.gz > $TMPDIR/$name.fa

    python3 $ROOT/apps/fcs.py screen genome \
        --fasta $TMPDIR/$name.fa  \
        --out-dir $TMPDIR/$name/ \
        --gx-db "$DB" \
        --tax-id 3635 # G. hirsutum

    tar -cf $DIR/1-fcs-gx/$name.tar -C $TMPDIR/$name/ \
        --transform="s#$name.3635.##" \
        $name.3635.{fcs_gx_report.txt,taxonomy.rpt}
done
